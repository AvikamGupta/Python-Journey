## ⚡ Python Syntax Showcase

```python
# 🔹 Basics
name = "Avikam"
print(f"Hello, {name}! Let's learn Python 🐍")

# 🔹 List Comprehension — Pythonic & Clean
squares = [x**2 for x in range(1, 11) if x % 2 == 0]
print(squares)  # [4, 16, 36, 64, 100]

# 🔹 Functions with Default Args
def greet(name, msg="Keep coding!"):
    return f"Hey {name}, {msg}"

# 🔹 OOP
class Student:
    def __init__(self, name, marks):
        self.name  = name
        self.marks = marks

    def grade(self):
        return "A" if self.marks >= 90 else "B" if self.marks >= 75 else "C"

s = Student("Avikam", 92)
print(f"{s.name} → Grade: {s.grade()}")

# 🔹 File Handling
with open("notes.txt", "r") as f:
    content = f.read()

# 🔹 Decorator
def logger(func):
    def wrapper(*args, **kwargs):
        print(f"Calling {func.__name__}...")
        return func(*args, **kwargs)
    return wrapper

@logger
def add(a, b):
    return a + b

# 🔹 Exception Handling
try:
    result = 10 / 0
except ZeroDivisionError as e:
    print(f"Error: {e}")
finally:
    print("Always runs.")
```
